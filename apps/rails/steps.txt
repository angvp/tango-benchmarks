# From an empty directory to a running pet shop with its staff account.
gem install rails -v 8.1.4
rails new . --api --database=postgresql --skip-solid --skip-docker --skip-kamal --skip-thruster
bundle add bcrypt
bin/rails generate migration CreateShop
