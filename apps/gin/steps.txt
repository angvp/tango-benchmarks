# From an empty directory to a running pet shop with its staff account.
go install github.com/sqlc-dev/sqlc/cmd/sqlc@v1.28.0
go mod init petshop
go get github.com/gin-gonic/gin github.com/jackc/pgx/v5 golang.org/x/crypto
sqlc generate
go build -o petshop .
GIN_MODE=release DATABASE_URL=postgres://bench:bench@localhost:5432/bench ./petshop
