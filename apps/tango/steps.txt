# From an empty directory to a running pet shop with a staff member who can log in to the admin.
go install github.com/angvp/tango/cmd/tango@v0.5.0
tango newproject petshop
tango makemigrations
# hand-edit: add ",index" to the three fk= tags in apps/shop/models.go
tango makemigrations
TANGO_DB_DSN=postgres://bench:bench@localhost:5432/bench tango migrate
TANGO_DB_DSN=postgres://bench:bench@localhost:5432/bench TANGO_ADMIN_PASSWORD=bench-staff-password tango admin create bench-staff
TANGO_DB_DSN=postgres://bench:bench@localhost:5432/bench go run .
